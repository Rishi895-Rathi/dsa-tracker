question number 2091. Removing Minimum and Maximum From Array
class Solution {
    public int minimumDeletions(int[] nums) {
        int n = nums.length;
        int minIndex = 0;
        int maxIndex = 0;
        for(int i = 0; i < n; i++){
            if(nums[i] < nums[minIndex]){
                minIndex = i;
            }
            if(nums[i] > nums[maxIndex]){
                maxIndex = i;
            }
        }

        int left = Math.min(minIndex, maxIndex);
        int right = Math.max(minIndex, maxIndex);

        int bothLeft = right + 1;
        int bothRight = n - left;
        int mixed = (left + 1) + (n - right);

        return Math.min(bothLeft, Math.min(bothRight, mixed));
    }
}

question number 4037. Maximum Valid Split Positions II
class Solution {
    int n;
    int[] pre, suf;
    int[][] sp;
    int[] lg;

    int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }

    int query(int l, int r) {
        int k = lg[r - l + 1];
        return gcd(sp[k][l], sp[k][r - (1 << k) + 1]);
    }

    int rangeGcd(int l, int r) {
        if (l > r) return 0;
        return query(l, r);
    }

    int preAt(int i) {
        return i < 0 ? 0 : pre[i];
    }

    int sufAt(int i) {
        return i >= n ? 0 : suf[i];
    }

    public int maxValidSplits(int[] nums) {
        n = nums.length;

        pre = new int[n];
        suf = new int[n];

        pre[0] = nums[0];
        for (int i = 1; i < n; i++) {
            pre[i] = gcd(pre[i - 1], nums[i]);
        }

        suf[n - 1] = nums[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            suf[i] = gcd(suf[i + 1], nums[i]);
        }

        lg = new int[n + 1];
        for (int i = 2; i <= n; i++) {
            lg[i] = lg[i / 2] + 1;
        }

        int K = lg[n] + 1;
        sp = new int[K][n];

        for (int i = 0; i < n; i++) {
            sp[0][i] = nums[i];
        }

        for (int k = 1; k < K; k++) {
            for (int i = 0; i + (1 << k) <= n; i++) {
                sp[k][i] = gcd(
                    sp[k - 1][i],
                    sp[k - 1][i + (1 << (k - 1))]
                );
            }
        }

        int best = 0;

        // Without removing any element
        for (int i = 0; i < n - 1; i++) {
            if (preAt(i) == sufAt(i + 1)) {
                best++;
            }
        }

        // Remove one element r
        for (int r = 0; r < n; r++) {
            int cnt = 0;

            // Split is before the removed element
            int lo = 1;
            int hi = r - 1;

            if (lo <= hi) {

                int a = lo;
                int b = hi;
                int c1 = hi + 1;

                while (a <= b) {
                    int m = a + (b - a) / 2;

                    int rightGcd = gcd(
                        rangeGcd(m, r - 1),
                        sufAt(r + 1)
                    );

                    if (preAt(m - 1) <= rightGcd) {
                        c1 = m;
                        b = m - 1;
                    } else {
                        a = m + 1;
                    }
                }

                a = lo;
                b = hi;
                int c2 = lo - 1;

                while (a <= b) {
                    int m = a + (b - a) / 2;

                    int rightGcd = gcd(
                        rangeGcd(m, r - 1),
                        sufAt(r + 1)
                    );

                    if (preAt(m - 1) >= rightGcd) {
                        c2 = m;
                        a = m + 1;
                    } else {
                        b = m - 1;
                    }
                }

                if (c1 <= c2) {
                    cnt += c2 - c1 + 1;
                }
            }

            // Split is after the removed element
            lo = r + 2;
            hi = n - 1;

            if (lo <= hi) {

                int a = lo;
                int b = hi;
                int c1 = hi + 1;

                while (a <= b) {
                    int m = a + (b - a) / 2;

                    int leftGcd = gcd(
                        preAt(r - 1),
                        rangeGcd(r + 1, m - 1)
                    );

                    if (leftGcd <= sufAt(m)) {
                        c1 = m;
                        b = m - 1;
                    } else {
                        a = m + 1;
                    }
                }

                a = lo;
                b = hi;
                int c2 = lo - 1;

                while (a <= b) {
                    int m = a + (b - a) / 2;

                    int leftGcd = gcd(
                        preAt(r - 1),
                        rangeGcd(r + 1, m - 1)
                    );

                    if (leftGcd >= sufAt(m)) {
                        c2 = m;
                        a = m + 1;
                    } else {
                        b = m - 1;
                    }
                }

                if (c1 <= c2) {
                    cnt += c2 - c1 + 1;
                }
            }

            // Split directly around the removed element
            if (r >= 1 && r <= n - 2) {
                if (preAt(r - 1) == sufAt(r + 1)) {
                    cnt++;
                }
            }

            best = Math.max(best, cnt);
        }

        return best;
    }
}

question number 4035. Maximum Valid Split Positions I
class Solution {
    public int maxValidSplits(int[] nums) {
        int n = nums.length;
        int ans = 0;
        for (int remove = -1; remove < n; remove++) {
            int m = n - (remove == -1 ? 0 : 1);
            if (m < 2)
                continue;
            int[] arr = new int[m];
            int k = 0;
            for (int i = 0; i < n; i++) {
                if (i == remove)
                    continue;
                arr[k++] = nums[i];
            }
            int[] prefix = new int[m];
            int[] suffix = new int[m];
            prefix[0] = arr[0];
            suffix[m - 1] = arr[m - 1];
            for (int i = 1; i < m; i++) {
                prefix[i] = gcd(prefix[i - 1], arr[i]);
            }
            for (int i = m - 2; i >= 0; i--) {
                suffix[i] = gcd(suffix[i + 1], arr[i]);
            }
            int score = 0;
            for (int i = 0; i < m - 1; i++) {
                if (prefix[i] == suffix[i + 1]) {
                    score++;
                }
            }
            ans = Math.max(ans, score);
        }
        return ans;
    }

    private int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}

class Solution {

        for (int i = 0; i < nums.length; i++) {
            int x = nums[i];
            StringBuilder sb = new StringBuilder();
            char ch = 'a';
            while (x > 0 && ch < 'z') {
                if (x % 2 == 1)
                    sb.append(ch);
                x /= 2;
                ch++;
            }
            while (x > 0) {
                sb.append('z');
                x--;
            }
            ans[i] = sb.reverse().toString();
        }
        return ans;
    }

quesrtion number 4034. Minimum Bishop Moves to Reach Target
class Solution {
    public int minBishopMoves(int[] source, int[] target) {
        int sr = source[0];
        int sc = source[1];
        int tr = target[0];
        int tc = target[1];
        if(sr == tr && sc == tc){
            return 0;
        }
        Queue<int[]> queue = new LinkedList<>();
        boolean[][] visited = new boolean[9][9];
        queue.offer(new int[]{sr, sc, 0});
        visited[sr][sc] = true;
        int[][] dirs = {
            {1, 1},
            {1, -1},
            {-1, 1},
            {-1, -1}
        };
        while(!queue.isEmpty()){
            int[] curr = queue.poll();
            int x = curr[0];
            int y = curr[1];
            int moves = curr[2];
            if(x == tr && y == tc){
                return moves;
            }
            for(int[] dir : dirs) {
                int nx = x + dir[0];
                int ny = y + dir[1];
                while(nx >= 1 && nx <= 8 && ny >= 1 && ny <= 8){
                    if(!visited[nx][ny]){
                        visited[nx][ny] = true;
                        queue.offer(new int[]{nx, ny, moves + 1});
                    }
                    nx += dir[0];
                    ny += dir[1];
                }
            }
        }
        return -1;
    }

class Solution {
    public boolean isValidBST(TreeNode root) {
        return check(root, Long.MIN_VALUE, Long.MAX_VALUE);
    }

    private boolean check(TreeNode root, long min, long max) {
        if (root == null) {
            return true;
        }

        if (root.val <= min || root.val >= max) {
            return false;
        }

        return check(root.left, min, root.val) &&
               check(root.right, root.val, max);
    }
}

question number 501. Find Mode in Binary Search Tree
class Solution {
    List<Integer> ans = new ArrayList<>();
    int maxcount = 0, count = 0;
    Integer prev = null;
    public int[] findMode(TreeNode root) {
        
        inorder(root);
        int[] res = new int[ans.size()];
        for(int i = 0; i < ans.size() ; i++){
            res[i] = ans.get(i);
        }
        return res;
    }

    private void inorder(TreeNode root){
        if(root == null){
            return;
        }
        inorder(root.left);

        if(prev == null){
            count = 1;
        }else if(root.val == prev){
            count++;
        }else{
            count = 1;
        }
        if(count > maxcount){
            ans.clear();
            maxcount = count;
            ans.add(root.val);
        }
        else if(count == maxcount){
            ans.add(root.val);
        }
        prev = root.val;

        inorder(root.right);
    }
}

question number 4031. Find All Numbers Disappeared in an Array II
class Solution {
    public List<List<Integer>> findDisappearedNumbers(int[] nums, int lower, int upper) {

        List<List<Integer>> res = new ArrayList<>();
        Set<Integer> set = new HashSet<>();
        for(int num : nums){
            set.add(num);
        }
        int i = lower;
        while(i <= upper){
            if(set.contains(i)){
                i++;
                continue;
            }
            int start = i;
            while (i <= upper && !set.contains(i)) {
                i++;
            }
            int end = i - 1;
            res.add(Arrays.asList(start, end));
        }
        return res;
    }
}

class Solution {
    public boolean isPalindromic(String s) {
        StringBuilder binary = new StringBuilder();
        for(char ch : s.toCharArray()){
            String bits = String.format("%8s",
                    Integer.toBinaryString((int) ch))
                    .replace(' ', '0');
            binary.append(bits);
        }
        int left = 0;
        int right = binary.length() - 1;
        while(left < right){
            if(binary.charAt(left) != binary.charAt(right)){
                return false;
            }
            left++;
            right--;
        }
        return true;
    }
}

class Solution {
    public TreeNode sortedListToBST(ListNode head) {
             
        return solve(head , null);
    }
    private TreeNode solve(ListNode head,ListNode tail){
        if(head == tail) return null;

        ListNode mid = middle(head , tail);
        TreeNode root = new TreeNode(mid.val);
        root.left = solve(head , mid);
        root.right = solve(mid.next , tail);

        return root;
    }
    private ListNode middle(ListNode head , ListNode tail){
        ListNode slow = head;
        ListNode fast = head; 

        while(fast != tail && fast.next != tail){
            fast = fast.next.next;
            slow = slow.next; 
        }
        return slow;
    }
}

question number 110 Balanced Binary Tree
class Solution {

    public boolean isBalanced(TreeNode root) {
        return height(root) != -1;
    }
    
    private int height(TreeNode node) {
        if(node == null){
            return 0;
        }
        int leftHeight = height(node.left);
        if(leftHeight == -1){
            return -1;
        }
        int rightHeight = height(node.right);
        if(rightHeight == -1){
            return -1;
        }
        if(Math.abs(leftHeight - rightHeight) > 1){
            return -1;
        }
        return Math.max(leftHeight, rightHeight) + 1;
    }
}

class Solution {
    public List<Integer> spiralOrder(int[][] mat) {
        List<Integer> res = new ArrayList<>();

        int m = mat.length , n = mat[0].length;
        int top = 0 , bottom = m - 1 , left = 0 , right = n - 1;
        while(top <= bottom && left <= right){
            for(int j = left; j <= right; j++){
                res.add(mat[top][j]);
            }
            top++;
            for(int i = top; i <= bottom; i++){
                res.add(mat[i][right]);
            }
            right--;
            if(top <= bottom){
                for(int j = right; j >= left; j--){
                    res.add(mat[bottom][j]);
                }
                bottom--;
            }
            if(left <= right) {
                for(int i = bottom; i >= top; i--){
                    res.add(mat[i][left]);
                }
                left++;
            }
        }
        return res;
    }
}

question number 3348. Smallest Divisible Digit Product II
class Solution {
    private static final int[][] FACTOR_OF_DIGIT = new int[10][4]; // [pow2, pow3, pow5, pow7]
    static {
        FACTOR_OF_DIGIT[2] = new int[]{1, 0, 0, 0};
        FACTOR_OF_DIGIT[3] = new int[]{0, 1, 0, 0};
        FACTOR_OF_DIGIT[4] = new int[]{2, 0, 0, 0};
        FACTOR_OF_DIGIT[5] = new int[]{0, 0, 1, 0};
        FACTOR_OF_DIGIT[6] = new int[]{1, 1, 0, 0};
        FACTOR_OF_DIGIT[7] = new int[]{0, 0, 0, 1};
        FACTOR_OF_DIGIT[8] = new int[]{3, 0, 0, 0};
        FACTOR_OF_DIGIT[9] = new int[]{0, 2, 0, 0};
    }

    public String smallestNumber(String num, long t) {
        int[] need = getPrimeCount(t);
        if (need == null) {
            return "-1";
        }

        int[] minFactors = getFactorCount(need);
        int minDigitsNeeded = sum(minFactors);
        int n = num.length();

        if (minDigitsNeeded > n) {
            return construct(minFactors);
        }

        int[] prefixCount = getPrimeCountFromString(num);
        int firstZeroIndex = num.indexOf('0');
        if (firstZeroIndex == -1) {
            firstZeroIndex = n;
            if (isSubset(need, prefixCount)) {
                return num;
            }
        }

        for (int i = n - 1; i >= 0; i--) {
            int d = num.charAt(i) - '0';
            prefixCount = subtract(prefixCount, FACTOR_OF_DIGIT[d]);
            int spaceAfter = n - 1 - i;

            if (i > firstZeroIndex) {
                continue;
            }

            for (int biggerDigit = d + 1; biggerDigit < 10; biggerDigit++) {
                int[] remaining = subtract(subtract(need, prefixCount), FACTOR_OF_DIGIT[biggerDigit]);
                int[] factorsAfterReplacement = getFactorCount(remaining);
                int needed = sum(factorsAfterReplacement);

                if (needed <= spaceAfter) {
                    int fillOnes = spaceAfter - needed;
                    StringBuilder sb = new StringBuilder();
                    sb.append(num, 0, i);
                    sb.append((char) ('0' + biggerDigit));
                    for (int k = 0; k < fillOnes; k++) {
                        sb.append('1');
                    }
                    sb.append(construct(factorsAfterReplacement));
                    return sb.toString();
                }
            }
        }

        int[] factorsExt = getFactorCount(need);
        int extLen = n + 1;
        int fillOnes = extLen - sum(factorsExt);
        StringBuilder sb = new StringBuilder();
        for (int k = 0; k < fillOnes; k++) {
            sb.append('1');
        }
        sb.append(construct(factorsExt));
        return sb.toString();
    }

    private int[] getPrimeCount(long t) {
        int[] count = new int[4];
        int[] primes = {2, 3, 5, 7};
        for (int p = 0; p < 4; p++) {
            while (t % primes[p] == 0) {
                t /= primes[p];
                count[p]++;
            }
        }
        if (t != 1) {
            return null;
        }
        return count;
    }

    private int[] getPrimeCountFromString(String num) {
        int[] count = new int[4];
        for (int i = 0; i < num.length(); i++) {
            int d = num.charAt(i) - '0';
            for (int p = 0; p < 4; p++) {
                count[p] += FACTOR_OF_DIGIT[d][p];
            }
        }
        return count;
    }

    // Given required (pow2,pow3,pow5,pow7), find the digit-count representation
    // that needs the FEWEST total digits (greedy: prefer 8,9 over 2,3,4 etc.)
    private int[] getFactorCount(int[] count) {
        int[] res = new int[10];
        int pow2 = count[0];
        int pow3 = count[1];

        int count8 = pow2 / 3;
        int remaining2 = pow2 % 3;
        int count9 = pow3 / 2;
        int count3 = pow3 % 2;
        int count4 = remaining2 / 2;
        int count2 = remaining2 % 2;
        int count6 = 0;

        if (count2 == 1 && count3 == 1) {
            count2 = 0;
            count3 = 0;
            count6 = 1;
        }
        if (count3 == 1 && count4 == 1) {
            count2 = 1;
            count6 = 1;
            count3 = 0;
            count4 = 0;
        }

        res[2] = count2;
        res[3] = count3;
        res[4] = count4;
        res[5] = count[2];
        res[6] = count6;
        res[7] = count[3];
        res[8] = count8;
        res[9] = count9;
        return res;
    }

    private String construct(int[] factors) {
        StringBuilder sb = new StringBuilder();
        for (int digit = 2; digit <= 9; digit++) {
            for (int k = 0; k < factors[digit]; k++) {
                sb.append((char) ('0' + digit));
            }
        }
        return sb.toString();
    }

    private boolean isSubset(int[] a, int[] b) {
        for (int i = 0; i < 4; i++) {
            if (b[i] < a[i]) {
                return false;
            }
        }
        return true;
    }

    private int[] subtract(int[] a, int[] b) {
        int[] res = new int[a.length];
        for (int i = 0; i < a.length; i++) {
            res[i] = Math.max(0, a[i] - b[i]);
        }
        return res;
    }

    private int sum(int[] arr) {
        int s = 0;
        for (int v : arr) {
            s += v;
        }
        return s;
    }
}

class Solution {
    public boolean isPalindrome(ListNode head) {        
        ListNode slow = head ;
        ListNode fast = head ;

        //now find the middle 
        while(fast != null && fast.next != null){
            fast = fast.next.next ;
            slow = slow.next ;
        }
        // now the linked list looks like 1 2 1 2 that is after the reversing from the middle 
        ListNode prev = null ;
        ListNode curr = slow ; 
        while(curr != null){
            ListNode next = curr.next ; 
            curr.next = prev ;
            prev = curr ;
            curr = next ;
        }
        //check wether it is palindorme or not 
        ListNode first = head;
        ListNode second = prev;
        while(second != null){
            if(first.val != second.val){
                return false;
            }
            first = first.next;
            second = second.next;
        }
        return true;
    }
}

        long second = Long.MIN_VALUE;
        long third = Long.MIN_VALUE;

        for (int num : nums) {
            if (num == first || num == second || num == third) {
                continue;
            }

            if (num > first) {
                third = second;
                second = first;
                first = num;
            } else if (num > second) {
                third = second;
                second = num;
            } else if (num > third) {
                third = num;
            }
        }

        if (third == Long.MIN_VALUE) {
            return (int) first;
        }

        return (int) third;
    }

class Solution {
    Integer[] dp;

    public String stoneGameIII(int[] stoneValue) {
        int diff = solve(stoneValue, 0);

        if (diff > 0) return "Alice";
        if (diff < 0) return "Bob";
        return "Tie";
    }

    private int solve(int[] stoneValue, int i) {
        if (i >= stoneValue.length) {
            return 0;
        }

        if (dp[i] != null) {
            return dp[i];
        }

        int take = 0;
        int ans = Integer.MIN_VALUE;

        for (int x = 1; x <= 3 && i + x - 1 < stoneValue.length; x++) {
            take += stoneValue[i + x - 1];
            ans = Math.max(ans, take - solve(stoneValue, i + x));
        }

        return dp[i] = ans;
    }
}

            return dp[i][j];
        }

        int left = piles[i] - solve(piles, i + 1, j);
        int right = piles[j] - solve(piles, i, j - 1);

        return dp[i][j] = Math.max(left, right);
    }

        dp = new Integer[n][n];

        int left = nums[i] - solve(nums, i + 1, j);

class Solution {
    public String smallestPalindrome(String s, int k) {
        int n = s.length();
        int[] freq = new int[26];
        for(char ch : s.toCharArray()){
            freq[ch - 'a']++;
        }
        int[] half = new int[26];
        char mid = 0;
        for(int i = 0; i < 26; i++){
            half[i] = freq[i] / 2;
            if(freq[i] % 2 == 1){
                mid = (char) (i + 'a');
            }
        }
        int halfLen = 0;
        for(int x : half) halfLen += x;
        if(halfLen == 0){
            return(k == 1 && mid != 0) ? String.valueOf(mid) : "";
        }
        long totalPerm = countArrangements(half, k);
        if(totalPerm < k) return "";
        StringBuilder left = new StringBuilder();
        for(int pos = 0; pos < halfLen; pos++){
            for(int c = 0; c < 26; c++){
                if(half[c] == 0) continue;
                half[c]--;
                long cnt = countArrangements(half, k);
                if(cnt >= k){
                    left.append((char) (c + 'a'));
                    break;
                }else{
                    half[c]++;
                    k -= cnt;
                }
            }
        }
        String leftStr = left.toString();
        String rightStr = left.reverse().toString();
        if(mid != 0){
            return leftStr + mid + rightStr;
        }else{
            return leftStr + rightStr;
        }
    }
    private long countArrangements(int[] counts, long limit){
        int total = 0;
        for (int x : counts) total += x;
        long res = 1;
        for(int x : counts){
            if(x == 0) continue;
            res = res * nCr(total, x, limit);
            if(res >= limit) return res; 
            total -= x;
        }
        return res;
    }
    private long nCr(int n, int r, long limit){
        if(r > n) return 0;
        r = Math.min(r, n - r);
        long ans = 1;
        for(int i = 1; i <= r; i++){
            ans = ans * (n - r + i) / i;
            if(ans >= limit) return ans;
        }
        return ans;
    }
}

class Solution {
    public int uniqueXorTriplets(int[] nums) {
        int mx = 0;
        for (int x : nums) {
            mx = Math.max(mx, x);
        }
        mx <<= 1; // 2 * max(nums), all XORs < 2*max

        boolean[] pairXorPresent = new boolean[mx];
        // Step 1: mark all possible a ^ b
        for (int a : nums) {
            for (int b : nums) {
                pairXorPresent[a ^ b] = true;
            }
        }

        int[] tripletXorPresent = new int[mx];
        // Step 2: from each possible (a ^ b), combine with any c
        for (int ab = 0; ab < mx; ab++) {
            if (pairXorPresent[ab]) {
                for (int c : nums) {
                    tripletXorPresent[ab ^ c] = 1;
                }
            }
        }

        int ans = 0;
        for (int v : tripletXorPresent) {
            ans += v;
        }
        return ans;
    }

        HashSet<Integer> pairXors = new HashSet<>();
        int n = nums.length;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {  // i,j independent (we allow repetition)
                pairXors.add(nums[i] ^ nums[j]);  // covers a^a, a^b, etc.
            }

        // Step 2: (a^b) ^ c for all c
        HashSet<Integer> tripletXors = new HashSet<>();
        for (int ab : pairXors) {
            for (int k = 0; k < n; k++) {
                tripletXors.add(ab ^ nums[k]);
            }
        }

        // Step 3: unique count
        return tripletXors.size();
    }

        return ans;

class Solution {
    public List<Integer> findMinHeightTrees(int n, int[][] edges) {
        if (n == 1) return Arrays.asList(0);
        List<Integer>[] graph = new ArrayList[n];
        for (int i = 0; i < n; i++) graph[i] = new ArrayList<>();
        int[] degree = new int[n];
        for (int[] edge : edges) {
            int u = edge[0], v = edge[1];
            graph[u].add(v);
            graph[v].add(u);
            degree[u]++;
            degree[v]++;
        }
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            if (degree[i] == 1) queue.offer(i);
        }
        int remaining = n;
        while (remaining > 2) {
            int size = queue.size();
            remaining -= size;
            while (size-- > 0) {
                int leaf = queue.poll();
                for (int neighbor : graph[leaf]) {
                    degree[neighbor]--;
                    if (degree[neighbor] == 1) queue.offer(neighbor);
                }
            }
        }
        return new ArrayList<>(queue);
    }
}

class Solution {
    public int[] pathExistenceQueries(int n, int[] nums, int maxDiff, int[][] queries) {

        for (int i = 0; i < n; i++) order[i] = i;
        Arrays.sort(order, (a, b) -> nums[a] - nums[b]);

        int[] sortedNums = new int[n];

            comp[i] = c;

        int r = 0;
        for (int i = 0; i < n; i++) {

            nextHop[i] = r;

        while ((1 << LOG) < n) LOG++;
        int[][] up = new int[LOG + 1][n];
        up[0] = nextHop;
        for (int k = 1; k <= LOG; k++) {
            for (int i = 0; i < n; i++) {
                up[k][i] = up[k - 1][up[k - 1][i]];
            }
        }

        int[] res = new int[queries.length];
        for (int q = 0; q < queries.length; q++) {
            int u = posInSorted[queries[q][0]];
            int v = posInSorted[queries[q][1]];
            if (u > v) { int t = u; u = v; v = t; }

            if (comp[u] != comp[v]) {
                res[q] = -1;
                continue;
            }

            int cur = u;
            int hops = 0;
            for (int k = LOG; k >= 0; k--) {
                if (up[k][cur] < v) {
                    cur = up[k][cur];
                    hops += (1 << k);
                }
            }
            if (cur != v) hops++;
            res[q] = hops;
        }
        return res;
    }
}

question number 3756. Concatenate Non-Zero Digits and Multiply by Sum II
class Solution {
    int Mod = 1000000007;
    public int[] sumAndMultiply(String s, int[][] queries) {
        int n = s.length();
        int[] res = new int[queries.length];
        long[] prefixSum = new long[n];
        long[] prefixNum = new long[n];
        int[] nonZeroCount = new int[n];
        long[] pow = new long[n + 1];

        // power of 10 array
        pow[0] = 1;
        for(int i = 1; i <= n; i++){
            pow[i] = (pow[i - 1] * 10) % Mod;
        }

        // base case: first character
        prefixSum[0] = s.charAt(0) - '0';
        prefixNum[0] = s.charAt(0) - '0';
        if(s.charAt(0) != '0'){
            nonZeroCount[0] = 1;
        }else{
            nonZeroCount[0] = 0;
        }

        // build prefix arrays
        for(int i = 1; i < n; i++){
            int d = s.charAt(i) - '0';

            prefixSum[i] = prefixSum[i - 1] + d;

            if(d == 0){
                prefixNum[i] = prefixNum[i - 1];
                nonZeroCount[i] = nonZeroCount[i - 1];
            }else{
                prefixNum[i] = (prefixNum[i - 1] * 10 + d) % Mod;
                nonZeroCount[i] = nonZeroCount[i - 1] + 1;
            }
        }

        // answer each query
        for(int i = 0; i < queries.length; i++){
            int l = queries[i][0];
            int r = queries[i][1];
            long prevNum;
            int prevCount;
            long prevSum;
            if(l == 0){
                prevNum = 0;
                prevCount = 0;
                prevSum = 0;
            }else{
                prevNum = prefixNum[l - 1];
                prevCount = nonZeroCount[l - 1];
                prevSum = prefixSum[l - 1];
            }
            int shiftCount = nonZeroCount[r] - prevCount;
            long x = prefixNum[r] - (prevNum * pow[shiftCount]) % Mod;
            x = (x + Mod) % Mod;
            long sum = prefixSum[r] - prevSum;
            long answer = (x * sum) % Mod;
            res[i] = (int) answer;
        }
        return res;
    }
}

class Solution {
    public long sumAndMultiply(int n) {
        long ans = 0;
        int div = 1 , sum = 0;
        while(n/div >= 10){
            div *= 10;
        }
        while(div > 0){
            int dig = n/div;
            if(dig != 0){
                ans = ans*10+dig;
                sum += dig;
            }
            n %= div;
            div /= 10;
        }
        return ans*sum;
    }
}

class Solution {
    public int removeCoveredIntervals(int[][] intervals) {
        int n = intervals.length;
        int ans = 0;
        for(int i = 0; i < n; i++){
            boolean covered = false;
            for(int j = 0; j < n; j++){
                if(i == j)
                    continue;
                if(intervals[j][0] <= intervals[i][0] && intervals[j][1] >= intervals[i][1]){
                    covered = true;
                    break;
                }
            }
            if(!covered)
                ans++;
        }
        return ans;
    }
}

                maxEnd = intervals[i][1];
            }
        }
        return ans;
    }

question number 1301 Number of paths with Max Score 
class Solution {

    int n;
    int MOD = 1000000007;

    class Pair {
        int score;
        int paths;

        Pair(int score, int paths) {
            this.score = score;
            this.paths = paths;
        }
    }

    Pair[][] dp;
    boolean[][] vis;

    int getIntFromChar(char ch) {
        if (ch == 'S' || ch == 'E')
            return 0;
        return ch - '0';
    }

    boolean isValid(int i, int j, List<String> board) {
        return i >= 0 && i < n &&
               j >= 0 && j < n &&
               board.get(i).charAt(j) != 'X';
    }

    Pair solve(int i, int j, List<String> board) {

        if (vis[i][j])
            return dp[i][j];

        char ch = board.get(i).charAt(j);

        if (ch == 'E') {
            vis[i][j] = true;
            dp[i][j] = new Pair(0, 1);
            return dp[i][j];
        }

        int upScore = Integer.MIN_VALUE;
        int leftScore = Integer.MIN_VALUE;
        int diagScore = Integer.MIN_VALUE;

        int upPaths = 0;
        int leftPaths = 0;
        int diagPaths = 0;

        // Up
        if (isValid(i - 1, j, board)) {

            Pair up = solve(i - 1, j, board);

            if (up.paths > 0) {
                upScore = up.score + getIntFromChar(ch);
                upPaths = up.paths;
            }
        }

        // Left
        if (isValid(i, j - 1, board)) {

            Pair left = solve(i, j - 1, board);

            if (left.paths > 0) {
                leftScore = left.score + getIntFromChar(ch);
                leftPaths = left.paths;
            }
        }

        // Diagonal
        if (isValid(i - 1, j - 1, board)) {

            Pair diag = solve(i - 1, j - 1, board);

            if (diag.paths > 0) {
                diagScore = diag.score + getIntFromChar(ch);
                diagPaths = diag.paths;
            }
        }

        int maxScore = Math.max(upScore, Math.max(leftScore, diagScore));

        int ways = 0;

        if (upScore == maxScore)
            ways = (ways + upPaths) % MOD;

        if (leftScore == maxScore)
            ways = (ways + leftPaths) % MOD;

        if (diagScore == maxScore)
            ways = (ways + diagPaths) % MOD;

        if (ways == 0)
            maxScore = 0;

        vis[i][j] = true;
        dp[i][j] = new Pair(maxScore, ways);

        return dp[i][j];
    }

    public int[] pathsWithMaxScore(List<String> board) {

        n = board.size();

        dp = new Pair[n][n];
        vis = new boolean[n][n];

        Pair ans = solve(n - 1, n - 1, board);

        if (ans.paths == 0)
            return new int[] {0, 0};

        return new int[] {ans.score, ans.paths};
    }
}

question number 2812. Find the Safest Path in a Grid

class Solution {
    int[][] dirs = {{0,1},{0,-1},{1,0},{-1,0}};
    
    public int maximumSafenessFactor(List<List<Integer>> grid) {
        int n = grid.size();
        int[][] dist = new int[n][n];
        
        for (int[] row : dist) Arrays.fill(row, -1);
        Deque<int[]> q = new ArrayDeque<>();
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid.get(i).get(j) == 1) {
                    q.offer(new int[]{i, j});
                    dist[i][j] = 0;
                }
            }
        }
        
        while (!q.isEmpty()) {
            int[] curr = q.poll();
            int x = curr[0], y = curr[1];
            for (int[] d : dirs) {
                int nx = x + d[0], ny = y + d[1];
                if (nx >= 0 && nx < n && ny >= 0 && ny < n && dist[nx][ny] == -1) {
                    dist[nx][ny] = dist[x][y] + 1;
                    q.offer(new int[]{nx, ny});
                }
            }
        }
        
        if (dist[0][0] == 0 || dist[n-1][n-1] == 0) return 0;
        int lo = 0, hi = n * 2;
        int ans = 0;
        
        while (lo <= hi) {
            int mid = lo + (hi - lo) / 2;
            if (canReach(dist, n, mid)) {
                ans = mid;
                lo = mid + 1;
            } else {
                hi = mid - 1;
            }
        }
        
        return ans;
    }
    
    private boolean canReach(int[][] dist, int n, int minSafe) {
        if (dist[0][0] < minSafe) return false;
        
        boolean[][] visited = new boolean[n][n];
        Deque<int[]> q = new ArrayDeque<>();
        q.offer(new int[]{0, 0});
        visited[0][0] = true;
        
        while (!q.isEmpty()) {
            int[] curr = q.poll();
            int x = curr[0], y = curr[1];
            
            if (x == n-1 && y == n-1) return true;
            
            for (int[] d : dirs) {
                int nx = x + d[0], ny = y + d[1];
                if (nx >= 0 && nx < n && ny >= 0 && ny < n
                    && !visited[nx][ny]
                    && dist[nx][ny] >= minSafe) {
                    visited[nx][ny] = true;
                    q.offer(new int[]{nx, ny});
                }
            }
        }
        return false;
    }
}

question number 3121. Count the Number of Special characters II
class Solution {
    public int numberOfSpecialChars(String word) {

        int lower = 0;
        int upper = 0;
        int invalid = 0;

        for (int i = 0; i < word.length(); i++) {

            char c = word.charAt(i);

            if (Character.isLowerCase(c)) {

                int bit = (1 << (c - 'a'));

                // checks the invalid char i.e. lowercase came before uppercase
                if ((upper & bit) != 0) {
                    invalid |= bit;
                }

                lower |= bit;

            } else {

                upper |= (1 << (c - 'A'));
            }
        }

        return Integer.bitCount((lower & upper & (~invalid)));
    }
}

class Solution {
    public int minElement(int[] nums) {
        
        int ans = Integer.MAX_VALUE;
        for(int num : nums){
            int temp = digitSum(num);
            if(ans > temp){
                ans = temp;
            }else{
                continue;
            }
        }
        return ans;
    }
    private int digitSum(int x){
        int sum = 0;
        while(x > 0){
            sum += x % 10;
            x /= 10;
        }
        return sum;
    }
}

class Solution {
    public int[] stringIndices(String[] wordsContainer, String[] wordsQuery) {
        TrieNode root = new TrieNode();
        for (int i = 0; i < wordsContainer.length; i++) {
            TrieNode node = root;
            String word = wordsContainer[i];
            if (node.bestIndex == -1 || word.length() < wordsContainer[node.bestIndex].length()) {
                node.bestIndex = i;
            }
            for (int j = word.length() - 1; j >= 0; j--) {
                int c = word.charAt(j) - 'a';
                if (node.children[c] == null) node.children[c] = new TrieNode();
                node = node.children[c];
                if (node.bestIndex == -1 || word.length() < wordsContainer[node.bestIndex].length()) {
                    node.bestIndex = i;
                }
            }
        }
        int[] result = new int[wordsQuery.length];
        for (int i = 0; i < wordsQuery.length; i++) {
            TrieNode node = root;
            String query = wordsQuery[i];
            for (int j = query.length() - 1; j >= 0; j--) {
                int c = query.charAt(j) - 'a';
                if (node.children[c] == null) break;
                node = node.children[c];
            }
            result[i] = node.bestIndex;
        }
        return result;
    }
    static class TrieNode {
        TrieNode[] children = new TrieNode[26];
        int bestIndex = -1;
    }

class Solution {

    public int earliestFinishTime(int[] landStartTime, int[] landDuration,
                                  int[] waterStartTime, int[] waterDuration) {

        int n = landStartTime.length;
        int m = waterStartTime.length;

        int ans = Integer.MAX_VALUE;

        // preprocess water rides
        int[][] water = new int[m][2];
        for (int i = 0; i < m; i++) {
            water[i][0] = waterStartTime[i];
            water[i][1] = waterDuration[i];
        }

        Arrays.sort(water, Comparator.comparingInt(a -> a[0]));

        int[] waterStarts = new int[m];
        int[] waterPrefixDur = new int[m];
        int[] waterSuffixFinish = new int[m];

        for (int i = 0; i < m; i++) {
            waterStarts[i] = water[i][0];
        }

        waterPrefixDur[0] = water[0][1];
        for (int i = 1; i < m; i++) {
            waterPrefixDur[i] = Math.min(waterPrefixDur[i - 1], water[i][1]);
        }

        waterSuffixFinish[m - 1] = water[m - 1][0] + water[m - 1][1];
        for (int i = m - 2; i >= 0; i--) {
            waterSuffixFinish[i] =
                Math.min(waterSuffixFinish[i + 1],
                         water[i][0] + water[i][1]);
        }

        // Land -> Water
        for (int i = 0; i < n; i++) {

            int landFinish = landStartTime[i] + landDuration[i];

            int idx = lowerBound(waterStarts, landFinish);

            int best = Integer.MAX_VALUE;

            if (idx < m)
                best = Math.min(best, waterSuffixFinish[idx]);

            if (idx > 0)
                best = Math.min(best,
                                landFinish + waterPrefixDur[idx - 1]);

            ans = Math.min(ans, best);
        }

        // preprocess land rides
        int[][] land = new int[n][2];
        for (int i = 0; i < n; i++) {
            land[i][0] = landStartTime[i];
            land[i][1] = landDuration[i];
        }

        Arrays.sort(land, Comparator.comparingInt(a -> a[0]));

        int[] landStarts = new int[n];
        int[] landPrefixDur = new int[n];
        int[] landSuffixFinish = new int[n];

        for (int i = 0; i < n; i++) {
            landStarts[i] = land[i][0];
        }

        landPrefixDur[0] = land[0][1];
        for (int i = 1; i < n; i++) {
            landPrefixDur[i] = Math.min(landPrefixDur[i - 1], land[i][1]);
        }

        landSuffixFinish[n - 1] = land[n - 1][0] + land[n - 1][1];
        for (int i = n - 2; i >= 0; i--) {
            landSuffixFinish[i] =
                Math.min(landSuffixFinish[i + 1],
                         land[i][0] + land[i][1]);
        }

        // Water -> Land
        for (int j = 0; j < m; j++) {

            int waterFinish = waterStartTime[j] + waterDuration[j];

            int idx = lowerBound(landStarts, waterFinish);

            int best = Integer.MAX_VALUE;

            if (idx < n)
                best = Math.min(best, landSuffixFinish[idx]);

            if (idx > 0)
                best = Math.min(best,
                                waterFinish + landPrefixDur[idx - 1]);

            ans = Math.min(ans, best);
        }

        return ans;
    }

    private int lowerBound(int[] arr, int target) {
        int l = 0, r = arr.length;

        while (l < r) {
            int mid = l + (r - l) / 2;

            if (arr[mid] < target)
                l = mid + 1;
            else
                r = mid;
        }

        return l;
    }

            graph[v].add(u);

            return 0;
        }

        return (int) modPow(2, maxDepth - 1);
    }

    private void dfs(int node, int parent, int depth,
                     List<Integer>[] graph) {

        maxDepth = Math.max(maxDepth, depth);

        for (int nei : graph[node]) {
            if (nei != parent) {
                dfs(nei, node, depth + 1, graph);
            }
        }
    }

    private long modPow(long base, long exp) {
        long ans = 1;

        while (exp > 0) {
            if ((exp & 1) == 1) {
                ans = (ans * base) % MOD;
            }

            base = (base * base) % MOD;
            exp >>= 1;
        }

        return ans;
    }

        // Number of edges in path = maxDepth
        // Each edge can be 1 or 2
        // Total assignments = 2^maxDepth
        // Half of them have odd sum
        // Answer = 2^(maxDepth - 1)

        if (maxDepth == 0) return 0;

        return modPow(2, maxDepth - 1);

    private int bfsMaxDepth(List<List<Integer>> graph, int n) {
        Queue<Integer> q = new LinkedList<>();
        boolean[] vis = new boolean[n + 1];

        q.offer(1);
        vis[1] = true;

        int depth = -1;

        while (!q.isEmpty()) {
            int size = q.size();
            depth++;

            for (int i = 0; i < size; i++) {
                int node = q.poll();

                for (int nei : graph.get(node)) {
                    if (!vis[nei]) {
                        vis[nei] = true;
                        q.offer(nei);

                }
            }
        }

class Solution {
    
    static final int MOD = 1_000_000_007;
    static final int LOG = 17;
    int[][] parent;
    int[] depth;
    List<Integer>[] graph;

    public int[] assignEdgeWeights(int[][] edges, int[][] queries) {
        int n = edges.length + 1;
        graph = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int[] e : edges) {
            int u = e[0], v = e[1];
            graph[u].add(v);
            graph[v].add(u);
        }
        parent = new int[LOG][n + 1];
        depth = new int[n + 1];
        dfs(1, -1);
        for (int k = 1; k < LOG; k++) {
            for (int v = 1; v <= n; v++) {
                if (parent[k - 1][v] != -1) {
                    parent[k][v] = parent[k - 1][parent[k - 1][v]];
                } else {
                    parent[k][v] = -1;
                }
            }
        }
        int[] ans = new int[queries.length];
        for (int i = 0; i < queries.length; i++) {
            int u = queries[i][0];
            int v = queries[i][1];
            if (u == v) {
                ans[i] = 0;
                continue;
            }
            int lca = lca(u, v);
            int d = depth[u] + depth[v] - 2 * depth[lca];
            ans[i] = modPow(2, d - 1);
        }
        return ans;
    }
    private void dfs(int u, int p) {
        parent[0][u] = p;
        for (int v : graph[u]) {
            if (v != p) {
                depth[v] = depth[u] + 1;
                dfs(v, u);
            }
        }
    }
    private int lca(int u, int v) {
        if (depth[u] < depth[v]) {
            int temp = u;
            u = v;
            v = temp;
        }
        for (int k = LOG - 1; k >= 0; k--) {
            if (parent[k][u] != -1 &&
                depth[parent[k][u]] >= depth[v]) {
                u = parent[k][u];
            }
        }
        if (u == v) return u;
        for (int k = LOG - 1; k >= 0; k--) {
            if (parent[k][u] != -1 &&
                parent[k][u] != parent[k][v]) {
                u = parent[k][u];
                v = parent[k][v];
            }
        }
        return parent[0][u];
    }
    private int modPow(long base, long exp) {
        long result = 1;
        while (exp > 0) {
            if ((exp & 1) == 1) {
                result = (result * base) % MOD;
            }
            base = (base * base) % MOD;
            exp >>= 1;
        }
        return (int) result;
    }

important notes to be taken 
ﾳ Lowest Common Ancestor (LCA)

LCA of two nodes is the lowest (deepest) node that is an ancestor of both nodes.

Why do we need LCA?
To find the distance between two nodes.
To answer path-related queries in trees.
Used in many tree problems on LeetCode and competitive programming.
Distance Formula
distance(u,v)=depth[u]+depth[v]−2×depth[LCA(u,v)]

2. What is LOG?

LOG represents the maximum power of 2 needed for jumps in Binary Lifting.
Why do we need it?

Instead of moving: we jump by power of 2

3. Binary Lifting 

a technique used to find the LCA quickly by jumping upward in power of 2

How LCA is Found?
Step 1

Bring both nodes to the same depth.

Step 2

Move both nodes upward together using the largest possible jumps.

Step 3

When their parents become the same, that parent is the LCA.

LCA = Lowest common ancestor of two nodes.

LOG = Number of power-of-two jump levels needed.

Binary Lifting = Store 2^k-th ancestors and use them to move up the tree in O(log n) time instead of O(n).

            if (c == '*') {
                len = Math.max(0, len - 1);

            }

                k = len - 1 - k;

            }
        }
        return dp[m - 1][n - 1];
    }

        int d = dir + 1;
        long count = 0;

class Solution {

        int n = nums.length;
        int ans = 0;
        for(int i = 0; i < n; i++){
            int count = 0;
            for(int j = i; j < n; j++){
                if(nums[j] == target){
                    count++;
                }
                int len = j - i + 1;
                if(count > len / 2){
                    ans++;
                }
            }
        }

questio number 3739 Count Subarray with Majority Element II
class Solution {
    public long countMajoritySubarrays(int[] nums, int target) {
        Map<Integer, Integer> map = new HashMap<>();
        map.put(0, 1);
        int cumSum = 0;
        long res = 0;
        long cntLess = 0;
        for (int num : nums) {
            int prev = cumSum;
            cumSum += (num == target) ? 1 : -1;
            if (cumSum > prev) {
                cntLess += map.getOrDefault(prev, 0);
            } else if (cumSum < prev) {
                cntLess -= map.getOrDefault(cumSum, 0);
            }
            res += cntLess;
            map.put(cumSum, map.getOrDefault(cumSum, 0) + 1);
        }
        return res;
    }
}

class Solution {
    public int maximumLength(int[] nums) {
        Map<Integer,Integer> map = new HashMap<>();
        int ans = 0;
        for(int i : nums){
           map.put(i, map.getOrDefault(i, 0) + 1);
        }

        if (map.containsKey(1)) {
            int ones = map.get(1);
            if (ones % 2 == 0)
                ans = ones - 1;
            else
                ans = ones;
        }

        for(int num : map.keySet()){
            if(nums == 1)continue ;
            int curr = num; 
            int len = 1; 
            while(map.containsKey(curr * curr) && map.get(curr) >= 2){
                curr = curr * curr;
                len += 2;
            }
            ans = Math.max(ans, len);
        }
        return ans;
    }
}
