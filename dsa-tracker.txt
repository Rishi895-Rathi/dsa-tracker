question number 3514 Number of Unique XOR Triplets II

accepted solution 
class Solution {
    public int uniqueXorTriplets(int[] nums) {
        int mx = 0;
        for (int x : nums) {
            mx = Math.max(mx, x);
        }
        mx <<= 1; // 2 * max(nums), all XORs < 2*max

        boolean[] pairXorPresent = new boolean[mx];
        // Step 1: mark all possible a ^ b
        for (int a : nums) {
            for (int b : nums) {
                pairXorPresent[a ^ b] = true;
            }
        }

        int[] tripletXorPresent = new int[mx];
        // Step 2: from each possible (a ^ b), combine with any c
        for (int ab = 0; ab < mx; ab++) {
            if (pairXorPresent[ab]) {
                for (int c : nums) {
                    tripletXorPresent[ab ^ c] = 1;
                }
            }
        }

        int ans = 0;
        for (int v : tripletXorPresent) {
            ans += v;
        }
        return ans;
    }
}

class Solution {
    public int uniqueXorTriplets(int[] nums) {
        // Step 1: all pair XORs
        HashSet<Integer> pairXors = new HashSet<>();
        int n = nums.length;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {  // i,j independent (we allow repetition)
                pairXors.add(nums[i] ^ nums[j]);  // covers a^a, a^b, etc.
            }
        }

        // Step 2: (a^b) ^ c for all c
        HashSet<Integer> tripletXors = new HashSet<>();
        for (int ab : pairXors) {
            for (int k = 0; k < n; k++) {
                tripletXors.add(ab ^ nums[k]);
            }
        }

        // Step 3: unique count
        return tripletXors.size();
    }
}

question number 3157 Smallest Palindromic Rearrangement I
class Solution {
    public String smallestPalindrome(String s) {
        int n = s.length();
        int[] freq = new int[26];
        for(char c : s.toCharArray()) freq[c - 'a']++;
        char[] ans = new char[n];
        int l = 0, r = n - 1;
        for(int i = 0; i < 26; i++) {
            char ch = (char)(i + 'a');
            while(freq[i] >= 2) {
                ans[l++] = ch;
                ans[r--] = ch;
                freq[i] -= 2;
            }
            if(freq[i] == 1) ans[n / 2] = ch;
        }
        return new String(ans);
    }
}

        int[] freq = new int[26];

        StringBuilder left = new StringBuilder();
        String mid = "";
        for(int i = 0; i < 26; i++) {
            if((freq[i] & 1) == 1) mid = String.valueOf((char)(i + 'a'));
            for(int j = 0; j < freq[i] / 2; j++) left.append((char)(i + 'a'));

        String right = new StringBuilder(left).reverse().toString();
        return left.toString() + mid + right;
    }

class Solution {
    public int uniqueXorTriplets(int[] nums) {
        int max = 0;

        for (int num : nums) {
            max = Math.max(max, num);
        }

        if (max <= 2)
            return max;

        int ans = 1;
        while (ans <= max)
            ans <<= 1;

        return ans;
    }

    Stack<Integer> minstack;

    public MinStack() {
        stack = new Stack<>();
        minstack = new Stack<>();
    }
    
    public void push(int value) {
        stack.push(value);
        if(minstack.isEmpty() || value <= minstack.peek()){
            minstack.push(value);
        }
    }
    
    public void pop() {
        if(minstack.peek().equals(stack.peek())){
            minstack.pop();
        }
        stack.pop();
    }
    
    public int top() {
        return stack.peek();
        
    }
    
    public int getMin() {
        return minstack.peek();   
    }
}

question number 3499 Maximize Active Section with Trade I
class Solution {
    public int maxActiveSectionsAfterTrade(String s) {
        int ones = 0;
        for (char c : s.toCharArray()) {
            if (c == '1') {
                ones++;
            }
        }

        s = "1" + s + "1";
        int n = s.length();

        List<Character> chars = new ArrayList<>();
        List<Integer> lens = new ArrayList<>();

        int i = 0;
        while (i < n) {
            char ch = s.charAt(i);
            int j = i;
            while (j < n && s.charAt(j) == ch) {
                j++;
            }
            chars.add(ch);
            lens.add(j - i);
            i = j;
        }

        int ans = ones;

        for (i = 2; i + 2 < chars.size(); i += 2) {
            int gain = lens.get(i - 1) + lens.get(i + 1);
            ans = Math.max(ans, ones + gain);
        }

        return ans;
    }
}

public class Solution {

    public int maxActiveSectionsAfterTrade(String input) {
        int totalNumberOfOnes = 0;
        int numberOfZerosBeforeOnes = 0;
        int maxNumberOfZerosToTransform = 0;

        for (int i = 0; i < input.length(); ++i) {

            int numberOfOnesBetweenZeros = 0;
            while (i < input.length() && input.charAt(i) == '1') {
                ++i;
                ++totalNumberOfOnes;
                ++numberOfOnesBetweenZeros;
            }

            int numberOfZerosAfterOnes = 0;
            while (i < input.length() && input.charAt(i) == '0') {
                ++i;
                ++numberOfZerosAfterOnes;
            }

            if (numberOfZerosBeforeOnes > 0 && numberOfOnesBetweenZeros > 0 && numberOfZerosAfterOnes > 0) {
                maxNumberOfZerosToTransform = Math.max(maxNumberOfZerosToTransform, numberOfZerosBeforeOnes + numberOfZerosAfterOnes);
            }

            --i;
            numberOfZerosBeforeOnes = numberOfZerosAfterOnes;
        }

        return totalNumberOfOnes + maxNumberOfZerosToTransform;
    }
}

question number Minimum Height Tree 
class Solution {
    public List<Integer> findMinHeightTrees(int n, int[][] edges) {
        if (n == 1) return Arrays.asList(0);
        List<Integer>[] graph = new ArrayList[n];
        for (int i = 0; i < n; i++) graph[i] = new ArrayList<>();
        int[] degree = new int[n];
        for (int[] edge : edges) {
            int u = edge[0], v = edge[1];
            graph[u].add(v);
            graph[v].add(u);
            degree[u]++;
            degree[v]++;
        }
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            if (degree[i] == 1) queue.offer(i);
        }
        int remaining = n;
        while (remaining > 2) {
            int size = queue.size();
            remaining -= size;
            while (size-- > 0) {
                int leaf = queue.poll();
                for (int neighbor : graph[leaf]) {
                    degree[neighbor]--;
                    if (degree[neighbor] == 1) queue.offer(neighbor);
                }
            }
        }
        return new ArrayList<>(queue);
    }
}

class Solution {
    public int[] findOrder(int n, int[][] prerequisites) {
        List<Integer>[] adj = new ArrayList[n];
        for(int i = 0; i < n; i++){
            adj[i] = new ArrayList<>();
        }
        int[] indegree = new int[n];
        for(int[] pair : prerequisites){
            int course = pair[0];
            int pre = pair[1];
            adj[pre].add(course);
            indegree[course]++;
        }
        Queue<Integer> queue = new LinkedList<>();
        for(int i = 0; i < n; i++){
            if(indegree[i] == 0){
                queue.offer(i);
            }
        }
        int[] order = new int[n];
        int index = 0;
        while(!queue.isEmpty()){
            int current = queue.poll();
            order[index++] = current;
            for(int next : adj[current]){
                indegree[next]--;
                if(indegree[next] == 0){
                    queue.offer(next);
                }
            }
        }
        if(index != n){
            return new int[0];
        }
        return order;
    }
}

class Solution {
    public boolean canFinish(int n, int[][] prerequisites) {
        List<Integer>[] adj = new ArrayList[n];
        for(int i = 0; i < n; i++){
            adj[i] = new ArrayList<>();
        }
        int[] indegree = new int[n];
        for(int[] pair : prerequisites){
            int course = pair[0];
            int pre = pair[1];
            adj[pre].add(course);
            indegree[course]++;
        }
        Queue<Integer> queue = new LinkedList<>();
        for(int i = 0; i < n; i++){
            if(indegree[i] == 0){
                queue.offer(i);
            }
        }
        int count = 0;
        while(!queue.isEmpty()){
            int current = queue.poll();
            count++;
            for(int next : adj[current]){
                indegree[next]--;
                if(indegree[next] == 0){
                    queue.offer(next);
                }
            }
        }

        return count == n;
    }
}

import java.util.Arrays;

class Solution {
    public int[] pathExistenceQueries(int n, int[] nums, int maxDiff, int[][] queries) {
        Integer[] order = new Integer[n];
        for (int i = 0; i < n; i++) order[i] = i;
        Arrays.sort(order, (a, b) -> nums[a] - nums[b]);

        int[] sortedNums = new int[n];
        int[] posInSorted = new int[n];
        for (int i = 0; i < n; i++) {
            sortedNums[i] = nums[order[i]];
            posInSorted[order[i]] = i;
        }

        int[] comp = new int[n];
        int c = 0;
        comp[0] = 0;
        for (int i = 1; i < n; i++) {
            if (sortedNums[i] - sortedNums[i - 1] > maxDiff) c++;
            comp[i] = c;
        }

        int[] nextHop = new int[n];
        int r = 0;
        for (int i = 0; i < n; i++) {
            if (r < i) r = i;
            while (r + 1 < n && sortedNums[r + 1] - sortedNums[i] <= maxDiff) r++;
            nextHop[i] = r;
        }

        int LOG = 1;
        while ((1 << LOG) < n) LOG++;
        int[][] up = new int[LOG + 1][n];
        up[0] = nextHop;
        for (int k = 1; k <= LOG; k++) {
            for (int i = 0; i < n; i++) {
                up[k][i] = up[k - 1][up[k - 1][i]];
            }
        }

        int[] res = new int[queries.length];
        for (int q = 0; q < queries.length; q++) {
            int u = posInSorted[queries[q][0]];
            int v = posInSorted[queries[q][1]];
            if (u > v) { int t = u; u = v; v = t; }

            if (comp[u] != comp[v]) {
                res[q] = -1;
                continue;
            }

            int cur = u;
            int hops = 0;
            for (int k = LOG; k >= 0; k--) {
                if (up[k][cur] < v) {
                    cur = up[k][cur];
                    hops += (1 << k);
                }
            }
            if (cur != v) hops++;
            res[q] = hops;
        }
        return res;
    }
}

question number 3756. Concatenate Non-Zero Digits and Multiply by Sum II
class Solution {
    int Mod = 1000000007;
    public int[] sumAndMultiply(String s, int[][] queries) {
        int n = s.length();
        int[] res = new int[queries.length];
        long[] prefixSum = new long[n];
        long[] prefixNum = new long[n];
        int[] nonZeroCount = new int[n];
        long[] pow = new long[n + 1];

        // power of 10 array
        pow[0] = 1;
        for(int i = 1; i <= n; i++){
            pow[i] = (pow[i - 1] * 10) % Mod;
        }

        // base case: first character
        prefixSum[0] = s.charAt(0) - '0';
        prefixNum[0] = s.charAt(0) - '0';
        if(s.charAt(0) != '0'){
            nonZeroCount[0] = 1;
        }else{
            nonZeroCount[0] = 0;
        }

        // build prefix arrays
        for(int i = 1; i < n; i++){
            int d = s.charAt(i) - '0';

            prefixSum[i] = prefixSum[i - 1] + d;

            if(d == 0){
                prefixNum[i] = prefixNum[i - 1];
                nonZeroCount[i] = nonZeroCount[i - 1];
            }else{
                prefixNum[i] = (prefixNum[i - 1] * 10 + d) % Mod;
                nonZeroCount[i] = nonZeroCount[i - 1] + 1;
            }
        }

        // answer each query
        for(int i = 0; i < queries.length; i++){
            int l = queries[i][0];
            int r = queries[i][1];
            long prevNum;
            int prevCount;
            long prevSum;
            if(l == 0){
                prevNum = 0;
                prevCount = 0;
                prevSum = 0;
            }else{
                prevNum = prefixNum[l - 1];
                prevCount = nonZeroCount[l - 1];
                prevSum = prefixSum[l - 1];
            }
            int shiftCount = nonZeroCount[r] - prevCount;
            long x = prefixNum[r] - (prevNum * pow[shiftCount]) % Mod;
            x = (x + Mod) % Mod;
            long sum = prefixSum[r] - prevSum;
            long answer = (x * sum) % Mod;
            res[i] = (int) answer;
        }
        return res;
    }
}

class Solution {
    public long sumAndMultiply(int n) {
        long ans = 0;
        int div = 1 , sum = 0;
        while(n/div >= 10){
            div *= 10;
        }
        while(div > 0){
            int dig = n/div;
            if(dig != 0){
                ans = ans*10+dig;
                sum += dig;
            }
            n %= div;
            div /= 10;
        }
        return ans*sum;
    }
}

class Solution {
    public long sumAndMultiply(int n) {
        int x = 0;
        int sum = 0;
        int i = 1;

        while (n > 0) {
            int m = n % 10;
            n /= 10;

            if (m == 0) continue;

            sum += m;
            x += m * i;
            i *= 10;
        }

        return (long) x * sum;
    }
}

question number 1288 Remove Covered intervals
//brute froce
class Solution {
    public int removeCoveredIntervals(int[][] intervals) {
        int n = intervals.length;
        int ans = 0;
        for(int i = 0; i < n; i++){
            boolean covered = false;
            for(int j = 0; j < n; j++){
                if(i == j)
                    continue;
                if(intervals[j][0] <= intervals[i][0] && intervals[j][1] >= intervals[i][1]){
                    covered = true;
                    break;
                }
            }
            if(!covered)
                ans++;
        }
        return ans;
    }
}

class Solution {
    public int removeCoveredIntervals(int[][] intervals) {
        Arrays.sort(intervals, (a, b) -> {
            if (a[0] == b[0]) {
                return Integer.compare(b[1], a[1]); // end descending
            }
            return Integer.compare(a[0], b[0]); // start ascending
        });
        int maxEnd = -1, ans = 0;
        for(int i = 0; i < intervals.length; i++){
            if(maxEnd < intervals[i][1]){
                ans++;
                maxEnd = intervals[i][1];
            }
        }
        return ans;
    }
}

class Solution {

    int n;
    int MOD = 1000000007;

    class Pair {
        int score;
        int paths;

        Pair(int score, int paths) {
            this.score = score;
            this.paths = paths;
        }
    }

    Pair[][] dp;
    boolean[][] vis;

    int getIntFromChar(char ch) {
        if (ch == 'S' || ch == 'E')
            return 0;
        return ch - '0';
    }

    boolean isValid(int i, int j, List<String> board) {
        return i >= 0 && i < n &&
               j >= 0 && j < n &&
               board.get(i).charAt(j) != 'X';
    }

    Pair solve(int i, int j, List<String> board) {

        if (vis[i][j])
            return dp[i][j];

        char ch = board.get(i).charAt(j);

        if (ch == 'E') {
            vis[i][j] = true;
            dp[i][j] = new Pair(0, 1);
            return dp[i][j];
        }

        int upScore = Integer.MIN_VALUE;
        int leftScore = Integer.MIN_VALUE;
        int diagScore = Integer.MIN_VALUE;

        int upPaths = 0;
        int leftPaths = 0;
        int diagPaths = 0;

        // Up
        if (isValid(i - 1, j, board)) {

            Pair up = solve(i - 1, j, board);

            if (up.paths > 0) {
                upScore = up.score + getIntFromChar(ch);
                upPaths = up.paths;
            }
        }

        // Left
        if (isValid(i, j - 1, board)) {

            Pair left = solve(i, j - 1, board);

            if (left.paths > 0) {
                leftScore = left.score + getIntFromChar(ch);
                leftPaths = left.paths;
            }
        }

        // Diagonal
        if (isValid(i - 1, j - 1, board)) {

            Pair diag = solve(i - 1, j - 1, board);

            if (diag.paths > 0) {
                diagScore = diag.score + getIntFromChar(ch);
                diagPaths = diag.paths;
            }
        }

        int maxScore = Math.max(upScore, Math.max(leftScore, diagScore));

        int ways = 0;

        if (upScore == maxScore)
            ways = (ways + upPaths) % MOD;

        if (leftScore == maxScore)
            ways = (ways + leftPaths) % MOD;

        if (diagScore == maxScore)
            ways = (ways + diagPaths) % MOD;

        if (ways == 0)
            maxScore = 0;

        vis[i][j] = true;
        dp[i][j] = new Pair(maxScore, ways);

        return dp[i][j];
    }

    public int[] pathsWithMaxScore(List<String> board) {

        n = board.size();

        dp = new Pair[n][n];
        vis = new boolean[n][n];

        Pair ans = solve(n - 1, n - 1, board);

        if (ans.paths == 0)
            return new int[] {0, 0};

        return new int[] {ans.score, ans.paths};
    }
}

question number 2812. Find the Safest Path in a Grid

class Solution {
    int[][] dirs = {{0,1},{0,-1},{1,0},{-1,0}};
    
    public int maximumSafenessFactor(List<List<Integer>> grid) {
        int n = grid.size();
        int[][] dist = new int[n][n];
        
        for (int[] row : dist) Arrays.fill(row, -1);
        Deque<int[]> q = new ArrayDeque<>();
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid.get(i).get(j) == 1) {
                    q.offer(new int[]{i, j});
                    dist[i][j] = 0;
                }
            }
        }
        
        while (!q.isEmpty()) {
            int[] curr = q.poll();
            int x = curr[0], y = curr[1];
            for (int[] d : dirs) {
                int nx = x + d[0], ny = y + d[1];
                if (nx >= 0 && nx < n && ny >= 0 && ny < n && dist[nx][ny] == -1) {
                    dist[nx][ny] = dist[x][y] + 1;
                    q.offer(new int[]{nx, ny});
                }
            }
        }
        
        if (dist[0][0] == 0 || dist[n-1][n-1] == 0) return 0;
        int lo = 0, hi = n * 2;
        int ans = 0;
        
        while (lo <= hi) {
            int mid = lo + (hi - lo) / 2;
            if (canReach(dist, n, mid)) {
                ans = mid;
                lo = mid + 1;
            } else {
                hi = mid - 1;
            }
        }
        
        return ans;
    }
    
    private boolean canReach(int[][] dist, int n, int minSafe) {
        if (dist[0][0] < minSafe) return false;
        
        boolean[][] visited = new boolean[n][n];
        Deque<int[]> q = new ArrayDeque<>();
        q.offer(new int[]{0, 0});
        visited[0][0] = true;
        
        while (!q.isEmpty()) {
            int[] curr = q.poll();
            int x = curr[0], y = curr[1];
            
            if (x == n-1 && y == n-1) return true;
            
            for (int[] d : dirs) {
                int nx = x + d[0], ny = y + d[1];
                if (nx >= 0 && nx < n && ny >= 0 && ny < n
                    && !visited[nx][ny]
                    && dist[nx][ny] >= minSafe) {
                    visited[nx][ny] = true;
                    q.offer(new int[]{nx, ny});
                }
            }
        }
        return false;
    }
}

class Solution {
    public int numberOfSpecialChars(String word) {

        int lower = 0;
        int upper = 0;
        int invalid = 0;

        for (int i = 0; i < word.length(); i++) {

            char c = word.charAt(i);

            if (Character.isLowerCase(c)) {

                int bit = (1 << (c - 'a'));

                // checks the invalid char i.e. lowercase came before uppercase
                if ((upper & bit) != 0) {
                    invalid |= bit;
                }

                lower |= bit;

            } else {

                upper |= (1 << (c - 'A'));
            }
        }

        return Integer.bitCount((lower & upper & (~invalid)));
    }
}

class Solution {

    public int earliestFinishTime(int[] landStartTime, int[] landDuration,
                                  int[] waterStartTime, int[] waterDuration) {

        int n = landStartTime.length;
        int m = waterStartTime.length;

        int ans = Integer.MAX_VALUE;

        // preprocess water rides
        int[][] water = new int[m][2];
        for (int i = 0; i < m; i++) {
            water[i][0] = waterStartTime[i];
            water[i][1] = waterDuration[i];
        }

        Arrays.sort(water, Comparator.comparingInt(a -> a[0]));

        int[] waterStarts = new int[m];
        int[] waterPrefixDur = new int[m];
        int[] waterSuffixFinish = new int[m];

        for (int i = 0; i < m; i++) {
            waterStarts[i] = water[i][0];
        }

        waterPrefixDur[0] = water[0][1];
        for (int i = 1; i < m; i++) {
            waterPrefixDur[i] = Math.min(waterPrefixDur[i - 1], water[i][1]);
        }

        waterSuffixFinish[m - 1] = water[m - 1][0] + water[m - 1][1];
        for (int i = m - 2; i >= 0; i--) {
            waterSuffixFinish[i] =
                Math.min(waterSuffixFinish[i + 1],
                         water[i][0] + water[i][1]);
        }

        // Land -> Water
        for (int i = 0; i < n; i++) {

            int landFinish = landStartTime[i] + landDuration[i];

            int idx = lowerBound(waterStarts, landFinish);

            int best = Integer.MAX_VALUE;

            if (idx < m)
                best = Math.min(best, waterSuffixFinish[idx]);

            if (idx > 0)
                best = Math.min(best,
                                landFinish + waterPrefixDur[idx - 1]);

            ans = Math.min(ans, best);
        }

        // preprocess land rides
        int[][] land = new int[n][2];
        for (int i = 0; i < n; i++) {
            land[i][0] = landStartTime[i];
            land[i][1] = landDuration[i];
        }

        Arrays.sort(land, Comparator.comparingInt(a -> a[0]));

        int[] landStarts = new int[n];
        int[] landPrefixDur = new int[n];
        int[] landSuffixFinish = new int[n];

        for (int i = 0; i < n; i++) {
            landStarts[i] = land[i][0];
        }

        landPrefixDur[0] = land[0][1];
        for (int i = 1; i < n; i++) {
            landPrefixDur[i] = Math.min(landPrefixDur[i - 1], land[i][1]);
        }

        landSuffixFinish[n - 1] = land[n - 1][0] + land[n - 1][1];
        for (int i = n - 2; i >= 0; i--) {
            landSuffixFinish[i] =
                Math.min(landSuffixFinish[i + 1],
                         land[i][0] + land[i][1]);
        }

        // Water -> Land
        for (int j = 0; j < m; j++) {

            int waterFinish = waterStartTime[j] + waterDuration[j];

            int idx = lowerBound(landStarts, waterFinish);

            int best = Integer.MAX_VALUE;

            if (idx < n)
                best = Math.min(best, landSuffixFinish[idx]);

            if (idx > 0)
                best = Math.min(best,
                                waterFinish + landPrefixDur[idx - 1]);

            ans = Math.min(ans, best);
        }

        return ans;
    }

    private int lowerBound(int[] arr, int target) {
        int l = 0, r = arr.length;

        while (l < r) {
            int mid = l + (r - l) / 2;

            if (arr[mid] < target)
                l = mid + 1;
            else
                r = mid;
        }

        return l;
    }

            base = (base * base) % MOD;
            exp >>= 1;
        }

        return ans;

class Solution {
    static final int MOD = 1_000_000_007;

    public int assignEdgeWeights(int[][] edges) {
        int n = edges.length + 1;

        // Find maximum depth from node 1
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int[] e : edges) {
            graph.get(e[0]).add(e[1]);
            graph.get(e[1]).add(e[0]);
        }

        int maxDepth = bfsMaxDepth(graph, n);

        // Number of edges in path = maxDepth
        // Each edge can be 1 or 2
        // Total assignments = 2^maxDepth
        // Half of them have odd sum
        // Answer = 2^(maxDepth - 1)

        if (maxDepth == 0) return 0;

        return modPow(2, maxDepth - 1);
    }

    private int bfsMaxDepth(List<List<Integer>> graph, int n) {
        Queue<Integer> q = new LinkedList<>();
        boolean[] vis = new boolean[n + 1];

        q.offer(1);
        vis[1] = true;

        int depth = -1;

        while (!q.isEmpty()) {
            int size = q.size();
            depth++;

            for (int i = 0; i < size; i++) {
                int node = q.poll();

                for (int nei : graph.get(node)) {
                    if (!vis[nei]) {
                        vis[nei] = true;
                        q.offer(nei);
                    }
                }
            }
        }

        return depth;
    }

    private int modPow(long base, long exp) {
        long res = 1;

        while (exp > 0) {
            if ((exp & 1) == 1) {
                res = (res * base) % MOD;
            }

            base = (base * base) % MOD;
            exp >>= 1;
        }

        return (int) res;
    }

question number 3559. Number of Ways to Assign Edge Weights II

class Solution {
    
    static final int MOD = 1_000_000_007;
    static final int LOG = 17;
    int[][] parent;
    int[] depth;
    List<Integer>[] graph;

    public int[] assignEdgeWeights(int[][] edges, int[][] queries) {
        int n = edges.length + 1;
        graph = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int[] e : edges) {
            int u = e[0], v = e[1];
            graph[u].add(v);
            graph[v].add(u);
        }
        parent = new int[LOG][n + 1];
        depth = new int[n + 1];
        dfs(1, -1);
        for (int k = 1; k < LOG; k++) {
            for (int v = 1; v <= n; v++) {
                if (parent[k - 1][v] != -1) {
                    parent[k][v] = parent[k - 1][parent[k - 1][v]];
                } else {
                    parent[k][v] = -1;
                }
            }
        }
        int[] ans = new int[queries.length];
        for (int i = 0; i < queries.length; i++) {
            int u = queries[i][0];
            int v = queries[i][1];
            if (u == v) {
                ans[i] = 0;
                continue;
            }
            int lca = lca(u, v);
            int d = depth[u] + depth[v] - 2 * depth[lca];
            ans[i] = modPow(2, d - 1);
        }
        return ans;
    }
    private void dfs(int u, int p) {
        parent[0][u] = p;
        for (int v : graph[u]) {
            if (v != p) {
                depth[v] = depth[u] + 1;
                dfs(v, u);
            }
        }
    }
    private int lca(int u, int v) {
        if (depth[u] < depth[v]) {
            int temp = u;
            u = v;
            v = temp;
        }
        for (int k = LOG - 1; k >= 0; k--) {
            if (parent[k][u] != -1 &&
                depth[parent[k][u]] >= depth[v]) {
                u = parent[k][u];
            }
        }
        if (u == v) return u;
        for (int k = LOG - 1; k >= 0; k--) {
            if (parent[k][u] != -1 &&
                parent[k][u] != parent[k][v]) {
                u = parent[k][u];
                v = parent[k][v];
            }
        }
        return parent[0][u];
    }
    private int modPow(long base, long exp) {
        long result = 1;
        while (exp > 0) {
            if ((exp & 1) == 1) {
                result = (result * base) % MOD;
            }
            base = (base * base) % MOD;
            exp >>= 1;
        }
        return (int) result;
    }

ﾳ Lowest Common Ancestor (LCA)

LCA of two nodes is the lowest (deepest) node that is an ancestor of both nodes.

Why do we need LCA?
To find the distance between two nodes.
To answer path-related queries in trees.
Used in many tree problems on LeetCode and competitive programming.
Distance Formula
distance(u,v)=depth[u]+depth[v]−2×depth[LCA(u,v)]

2. What is LOG?

LOG represents the maximum power of 2 needed for jumps in Binary Lifting.
Why do we need it?

Instead of moving: we jump by power of 2

3. Binary Lifting 

a technique used to find the LCA quickly by jumping upward in power of 2

How LCA is Found?
Step 1

Bring both nodes to the same depth.

Step 2

Move both nodes upward together using the largest possible jumps.

Step 3

When their parents become the same, that parent is the LCA.

questio number 3614. Process String with Special Operations II
class Solution {
    public char processStr(String s, long k) {
        long len = 0;
        for (char c : s.toCharArray()) {
            if (c == '*') {
                len = Math.max(0, len - 1);
            }else if (c == '#') {
                len <<= 1;
            }else if (c != '%') {
                len++;
            }
        }
        if (k >= len) {
            return '.';
        }
        for (int i = s.length() - 1; ; i--) {
            char c = s.charAt(i);

            if (c == '*') {
                len++;
            } 
            else if (c == '#') {
                len /= 2;
                if (k >= len) {
                    k -= len;
                }
            } 
            else if (c == '%') {
                k = len - 1 - k;
            } 
            else {
                len--;
                if (k == len) {
                    return c;
                }
            }
        }
    }
}

            list.get(i)[1] = Math.min(list.get(i)[1], list.get(i-1)[1] + distance);
        }

class Solution {
    public int uniquePaths(int m, int n) {
        int [][] dp = new int[m][n];
        for(int i = 0; i < m; i++){
            dp[i][0] = 1;
        }
        for(int i = 0; i < n; i++){
            dp[0][i] = 1;
        }

        for(int i = 1; i < m; i++){
            for(int j = 1; j < n; j++){
                dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
            }
        }
        return dp[m - 1][n - 1];
    }
}

questio number 3739 Count Subarray with Majority Element II
class Solution {
    public long countMajoritySubarrays(int[] nums, int target) {
        Map<Integer, Integer> map = new HashMap<>();
        map.put(0, 1);
        int cumSum = 0;
        long res = 0;
        long cntLess = 0;
        for (int num : nums) {
            int prev = cumSum;
            cumSum += (num == target) ? 1 : -1;
            if (cumSum > prev) {
                cntLess += map.getOrDefault(prev, 0);
            } else if (cumSum < prev) {
                cntLess -= map.getOrDefault(cumSum, 0);
            }
            res += cntLess;
            map.put(cumSum, map.getOrDefault(cumSum, 0) + 1);
        }
        return res;
    }
}

           map.put(i, map.getOrDefault(i, 0) + 1);
