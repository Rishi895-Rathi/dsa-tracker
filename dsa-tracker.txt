question number 3120. Count the Number of Special Characters I 
class Solution {
    public int numberOfSpecialChars(String word) {
        Set<Character> chars = new HashSet<>();
        for (char c : word.toCharArray()) chars.add(c);
        int count = 0;
        for (char c = 'a'; c <= 'z'; c++) {
            if (chars.contains(c) && chars.contains(Character.toUpperCase(c))){
                count++;
            }
        }
        return count;
    }
}

question number 3121. Count the Number of Special characters II
class Solution {
    public int numberOfSpecialChars(String word) {

        int lower = 0;
        int upper = 0;
        int invalid = 0;

        for (int i = 0; i < word.length(); i++) {

            char c = word.charAt(i);

            if (Character.isLowerCase(c)) {

                int bit = (1 << (c - 'a'));

                // checks the invalid char i.e. lowercase came before uppercase
                if ((upper & bit) != 0) {
                    invalid |= bit;
                }

                lower |= bit;

            } else {

                upper |= (1 << (c - 'A'));
            }
        }

        return Integer.bitCount((lower & upper & (~invalid)));
    }
}

question number 2540. minimum common value
class Solution {
    public int getCommon(int[] nums1, int[] nums2) {
        //with binary search
	for (int num : nums1) {
            int lo = 0, hi = nums2.length - 1;
            while (lo <= hi) {
                int mid = lo + (hi - lo) / 2;
                if (nums2[mid] == num) return num;
                else if (nums2[mid] < num) lo = mid + 1;
                else hi = mid - 1;
            }
        }
        return -1;
    }
}

question number 3300. minimum element after replacement with digit sum
class Solution {
    public int minElement(int[] nums) {
        
        int ans = Integer.MAX_VALUE;
        for(int num : nums){
            int temp = digitSum(num);
            if(ans > temp){
                ans = temp;
            }else{
                continue;
            }
        }
        return ans;
    }
    private int digitSum(int x){
        int sum = 0;
        while(x > 0){
            sum += x % 10;
            x /= 10;
        }
        return sum;
    }
}

question number 3093. Longest Common Suffix Queries
class Solution {
    public int[] stringIndices(String[] wordsContainer, String[] wordsQuery) {
        TrieNode root = new TrieNode();
        for (int i = 0; i < wordsContainer.length; i++) {
            TrieNode node = root;
            String word = wordsContainer[i];
            if (node.bestIndex == -1 || word.length() < wordsContainer[node.bestIndex].length()) {
                node.bestIndex = i;
            }
            for (int j = word.length() - 1; j >= 0; j--) {
                int c = word.charAt(j) - 'a';
                if (node.children[c] == null) node.children[c] = new TrieNode();
                node = node.children[c];
                if (node.bestIndex == -1 || word.length() < wordsContainer[node.bestIndex].length()) {
                    node.bestIndex = i;
                }
            }
        }
        int[] result = new int[wordsQuery.length];
        for (int i = 0; i < wordsQuery.length; i++) {
            TrieNode node = root;
            String query = wordsQuery[i];
            for (int j = query.length() - 1; j >= 0; j--) {
                int c = query.charAt(j) - 'a';
                if (node.children[c] == null) break;
                node = node.children[c];
            }
            result[i] = node.bestIndex;
        }
        return result;
    }
    static class TrieNode {
        TrieNode[] children = new TrieNode[26];
        int bestIndex = -1;
    }
}

class Solution {
    public boolean asteroidsDestroyed(int mass, int[] asteroids) {
        Arrays.sort(asteroids);
        long mass1 = mass;
        for(int i = 0; i < asteroids.length; i++){

question number 2553 Seoerate the Digit in an Array
class Solution {
    public int[] separateDigits(int[] nums) {
        
        ArrayList<Integer> list = new ArrayList<>();
        for(int num : nums){
            int divisor = 1;
            while(num / divisor >= 10) {
                divisor *= 10;
            }
            while(divisor > 0) {
                int digit = num / divisor;
                list.add(digit);
                num %= divisor;
                divisor /= 10;
            }
        }
        int[] arr = new int[list.size()];
        for (int i = 0; i < list.size(); i++) {
            arr[i] = list.get(i);
        }
        return arr;
    }
}

question number 3633. Earliest Finish Time for Land and Water Rides I
class Solution {
    public int earliestFinishTime(int[] landStartTime, int[] landDuration,
                                  int[] waterStartTime, int[] waterDuration) {

        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < landStartTime.length; i++) {
            for (int j = 0; j < waterStartTime.length; j++) {
                // land to water plan a
                int landFinish = landStartTime[i] + landDuration[i];
                int waterStart = Math.max(landFinish,waterStartTime[j]);
                int finish1 = waterStart + waterDuration[j];
                //water to land plan b
                int waterFinish = waterStartTime[j] + waterDuration[j];
                int landStart = Math.max(waterFinish,landStartTime[i]);
                int finish2 = landStart + landDuration[i];
                ans = Math.min(ans, finish1);
                ans = Math.min(ans, finish2);
            }
        }
        return ans;
    }
}

question number 3635 Earliest Finish Time for Land and Water RIdes II

class Solution {

    public int earliestFinishTime(int[] landStartTime, int[] landDuration,
                                  int[] waterStartTime, int[] waterDuration) {

        int n = landStartTime.length;
        int m = waterStartTime.length;

        int ans = Integer.MAX_VALUE;

        // preprocess water rides
        int[][] water = new int[m][2];
        for (int i = 0; i < m; i++) {
            water[i][0] = waterStartTime[i];
            water[i][1] = waterDuration[i];
        }

        Arrays.sort(water, Comparator.comparingInt(a -> a[0]));

        int[] waterStarts = new int[m];
        int[] waterPrefixDur = new int[m];
        int[] waterSuffixFinish = new int[m];

        for (int i = 0; i < m; i++) {
            waterStarts[i] = water[i][0];
        }

        waterPrefixDur[0] = water[0][1];
        for (int i = 1; i < m; i++) {
            waterPrefixDur[i] = Math.min(waterPrefixDur[i - 1], water[i][1]);
        }

        waterSuffixFinish[m - 1] = water[m - 1][0] + water[m - 1][1];
        for (int i = m - 2; i >= 0; i--) {
            waterSuffixFinish[i] =
                Math.min(waterSuffixFinish[i + 1],
                         water[i][0] + water[i][1]);
        }

        // Land -> Water
        for (int i = 0; i < n; i++) {

            int landFinish = landStartTime[i] + landDuration[i];

            int idx = lowerBound(waterStarts, landFinish);

            int best = Integer.MAX_VALUE;

            if (idx < m)
                best = Math.min(best, waterSuffixFinish[idx]);

            if (idx > 0)
                best = Math.min(best,
                                landFinish + waterPrefixDur[idx - 1]);

            ans = Math.min(ans, best);
        }

        // preprocess land rides
        int[][] land = new int[n][2];
        for (int i = 0; i < n; i++) {
            land[i][0] = landStartTime[i];
            land[i][1] = landDuration[i];
        }

        Arrays.sort(land, Comparator.comparingInt(a -> a[0]));

        int[] landStarts = new int[n];
        int[] landPrefixDur = new int[n];
        int[] landSuffixFinish = new int[n];

        for (int i = 0; i < n; i++) {
            landStarts[i] = land[i][0];
        }

        landPrefixDur[0] = land[0][1];
        for (int i = 1; i < n; i++) {
            landPrefixDur[i] = Math.min(landPrefixDur[i - 1], land[i][1]);
        }

        landSuffixFinish[n - 1] = land[n - 1][0] + land[n - 1][1];
        for (int i = n - 2; i >= 0; i--) {
            landSuffixFinish[i] =
                Math.min(landSuffixFinish[i + 1],
                         land[i][0] + land[i][1]);
        }

        // Water -> Land
        for (int j = 0; j < m; j++) {

            int waterFinish = waterStartTime[j] + waterDuration[j];

            int idx = lowerBound(landStarts, waterFinish);

            int best = Integer.MAX_VALUE;

            if (idx < n)
                best = Math.min(best, landSuffixFinish[idx]);

            if (idx > 0)
                best = Math.min(best,
                                waterFinish + landPrefixDur[idx - 1]);

            ans = Math.min(ans, best);
        }

        return ans;
    }

    private int lowerBound(int[] arr, int target) {
        int l = 0, r = arr.length;

        while (l < r) {
            int mid = l + (r - l) / 2;

            if (arr[mid] < target)
                l = mid + 1;
            else
                r = mid;
        }

        return l;
    }
}

question number 3751 total Waviness of number in Range I
class Solution {
    public int totalWaviness(int num1, int num2) {

        int ans = 0;
        while(num1 <= num2){
            int temp = num1, prev = 0;
            String s = String.valueOf(num1);
            prev = s.charAt(0)-'0';
            for (int i = 1; i < s.length()-1; i++){
                int dig = s.charAt(i) - '0';
                int nex = s.charAt(i+1)-'0';
                if((dig < prev && dig < nex) || (dig > prev && dig > nex)){
                    ans++;
                }
                prev = dig;
            }
            num1++;
        }
        return ans;
    }
}
