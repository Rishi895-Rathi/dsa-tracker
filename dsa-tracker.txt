question number 3120. Count the Number of Special Characters I 
class Solution {
    public int numberOfSpecialChars(String word) {
        Set<Character> chars = new HashSet<>();
        for (char c : word.toCharArray()) chars.add(c);
        int count = 0;
        for (char c = 'a'; c <= 'z'; c++) {
            if (chars.contains(c) && chars.contains(Character.toUpperCase(c))){
                count++;
            }
        }
        return count;
    }
}

question number 3121. Count the Number of Special characters II
class Solution {
    public int numberOfSpecialChars(String word) {

        int lower = 0;
        int upper = 0;
        int invalid = 0;

        for (int i = 0; i < word.length(); i++) {

            char c = word.charAt(i);

            if (Character.isLowerCase(c)) {

                int bit = (1 << (c - 'a'));

                // checks the invalid char i.e. lowercase came before uppercase
                if ((upper & bit) != 0) {
                    invalid |= bit;
                }

                lower |= bit;

            } else {

                upper |= (1 << (c - 'A'));
            }
        }

        return Integer.bitCount((lower & upper & (~invalid)));
    }
}

class Solution {
    public int getCommon(int[] nums1, int[] nums2) {
        //with binary search
	for (int num : nums1) {
            int lo = 0, hi = nums2.length - 1;
            while (lo <= hi) {
                int mid = lo + (hi - lo) / 2;
                if (nums2[mid] == num) return num;
                else if (nums2[mid] < num) lo = mid + 1;
                else hi = mid - 1;
            }
        }
        return -1;
    }
}
